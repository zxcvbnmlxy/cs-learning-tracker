若一个数（首位不为零）从左向右读与从右向左读都一样，我们就将其称为回文数。
例如，给定一个十进制数 56，将 56 加 65（即把 56 从右向左读），得到的 121 是一个回文数。

关键点：熟悉不同进制的数加法，懂得int与string之间的灵活转化


代码：
#include<bits/stdc++.h>
using namespace std;
int N;
string M;
char valTochar(int n){
    if(n>=10)return n-10+'A';
    else return n+'0';
}
int charToval(char c){
    if(c>='A')return c-'A'+10;
    else return c-'0';
}
bool is_huiwen(string s){
    int l=0,r=s.size()-1;
    while(l<=r){
        if(s[l]!=s[r])return false;
        l++,r--;
    }
    return true;
}

//不同进制之间的加法，模板                  重点

string calc(string s1,string s2){
    int len1=s1.size()-1,len2=s2.size()-1;
    int carry=0;
    string res;
    while(len1>=0||len2>=0){
        int a=(len1>=0)?charToval(s1[len1--]):0;
        int b=(len2>=0)?charToval(s2[len2--]):0;
        int sum=a+b+carry;
        int digit=sum%N;
        carry=sum/N;
        res+=valTochar(digit);
    }
    if(carry>0)res+=valTochar(carry);
    reverse(res.begin(),res.end());
    return res;
}

int main(){
    cin>>N>>M;
    if(is_huiwen(M)){cout<<0;return 0;}
    for(int step=1;step<=30;step++){
        string pre=M;
        reverse(M.begin(),M.end());
        M=calc(M,pre);
        if(is_huiwen(M)){cout<<"STEP="<<step;return 0;}
    }
    cout<<"Impossible!";
    return 0;
}
