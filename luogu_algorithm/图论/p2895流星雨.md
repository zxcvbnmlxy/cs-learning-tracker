2. 建模思路
1 建立mp[][]，初始值设为 INF（代表永久安全）
2 每颗流星：中心点 + 四邻，执行 
   mp[nx][ny] = min(mp[nx][ny], t);
3 BFS 状态：(x,y,当前时刻t)
4 通行条件：到达该点的时间 严格小于 格子摧毁时间
    t+1 < mp[nx][ny]
5 如果走到 mp[x][y]==INF，直接输出当前 t，程序结束


#include<bits/stdc++.h>
using namespace std;
vector<pair<int,int>>dir={{0,-1},{-1,0},{0,1},{1,0}};
int main(){
    int M;cin>>M;
    vector<vector<int>>mp(405,vector<int>(405,1001));
    vector<vector<int>>vis(405,vector<int>(405,0));
    for(int i=0;i<M;i++){
            int x,y,t;
            cin>>x>>y>>t;
            mp[x][y]=min(t,mp[x][y]);
            for(auto [dx,dy]:dir){
                int X=dx+x;
                int Y=dy+y;
                if(X>=0&&X<405&&Y>=0&&Y<405){
                    mp[X][Y]=min(t,mp[X][Y]);
            }
        }
    }
    if(mp[0][0]<=0){
        cout<<-1;return 0;
    }
    queue<tuple<int,int,int>>q;
    q.push({0,0,0});
    vis[0][0]=1;
    while(!q.empty()){
        auto [x,y,t]=q.front();
        q.pop();
        if(mp[x][y]==1001){
            cout<<t;return 0;
        }
        for(auto [dx,dy]:dir){
            int nx=dx+x;
            int ny=dy+y;
            if(nx>=0&&ny>=0&&ny<405&&nx<405&&t+1<mp[nx][ny]&&vis[nx][ny]==0){
                q.push({nx,ny,t+1});
                vis[nx][ny]=1;
            }
        }
    }
    cout<<-1;
    return 0;
}
