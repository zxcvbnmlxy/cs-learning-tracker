
本题求解字符串通过替换规则变换到目标串的最少步数，采用广度优先搜索（BFS）实现。以字符串作为搜索状态，队列保存当前字符串与变换步数，利用哈希集合记录已出现的字符串，避免重复搜索造成死循环。依次枚举每条替换规则，查找当前字符串内所有可匹配的子串，生成替换后的新状态。题目限定最多变换 10 步，对达到步数上限的分支进行剪枝。搜索过程中一旦得到目标字符串，立刻输出当前步数；全部路径遍历完毕仍未找到目标，则输出无解。BFS 能够保证首次搜到目标时得到最少变换次数，符合最短路求解需求。

用匹配规则中的A[i]长度然后枚举截取s中的字符串若有能够和A[i]匹配的，拼接在一起，然后记录没出现过的temp

substr(pos,len);pos+len<=s.size();
代码：

#include<bits/stdc++.h>
#include<string>
using namespace std;
int main(){
    vector<string>A;
    vector<string>B;
    unordered_set<string>vis;
    string a;
    int i=0;
    while(cin>>a){
        if(i%2==0)A.push_back(a);
        else B.push_back(a);
        i++;
    }
    queue<pair<string,int>>q;
    q.push({A[0],0});
    vis.emplace(A[0]);
    while(!q.empty()){
        auto [s,dist]=q.front();
        q.pop();
        if(dist>=10)continue;
        if(s==B[0]){
            cout<<dist;
            return 0;
        }
        for(int k=1;k<(int)A.size();k++){
            string ori=A[k];
            string dst=B[k];
            int len=ori.size();
            for(int i=0;i+len<=(int)s.size();i++){
                if(s.substr(i,len)==ori){
                    string temp;
                    temp= s.substr(0,i) + dst + s.substr(i+len,s.size()-len-i);
                    if(vis.count(temp))continue;
                    vis.emplace(temp);
                    q.push({temp,dist+1});
                }
            }
        }
    }
    cout<<"NO ANSWER!";
    return 0;
}
