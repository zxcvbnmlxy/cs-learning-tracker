#include<bits/stdc++.h>
using namespace std;

int main(){
    int n;
    vector<int>in;
    queue<int>q;
    cin>>n;
    vector<vector<int>>g(n+1);
    vector<int>len(n+1);
    vector<int>time(n+1,0);
    in.assign(n+1,0);
    for(int i=1;i<=n;i++){
        int id,t,pre;
        cin>>id>>t;
        len[id]=t;
        time[id]=t;
        while(cin>>pre&&pre!=0){
            g[pre].push_back(id);
            in[id]++;
        }
    }
    for(int i=1;i<=n;i++){
        if(in[i]==0)q.push(i);
    }
    while(!q.empty()){
        int u=q.front();
        q.pop();
        for(auto &v:g[u] ){
            time[v]=max(time[v],time[u]+len[v]);
            in[v]--;
            if(in[v]==0)q.push(v);
        }
    }
    cout<<*max_element(time.begin()+1,time.end())<<"\n";
    return 0;

}
