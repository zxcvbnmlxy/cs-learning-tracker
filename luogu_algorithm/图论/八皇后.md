#include<bits/stdc++.h>
using namespace std;
int cnt=0;
int n;
bool col[20]={false},dg[40]={false},udg[40]={false};
int graph[20],flag=0;
void dfs(int row){
    if(row>n){
        cnt++;
        if(cnt<=3){
            for(int i=1;i<=n;i++){
                cout<<graph[i]<<((i==n)?"\n":" ");
            }
        }
        return ;        
    }
    for(int j=1;j<=n;j++){
        if(!col[j]&&!dg[row-j+n]&&!udg[j+row]){
            graph[row]=j;
            col[j]=dg[row-j+n]=udg[j+row]=true;
            dfs(row+1);
            col[j]=dg[row-j+n]=udg[j+row]=false;
        }
    }
}
int main(){
    cin>>n;
    dfs(1);
    cout<<cnt;
    return 0;
}
