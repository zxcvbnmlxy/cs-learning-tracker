#include<bits/stdc++.h>
using namespace std;
vector<int>fa;
vector<int>sz;
//找根函数
int find(int x){
    if(fa[x]!=x)fa[x]=find(fa[x]);//沿着父亲找，压缩路径
    return fa[x];
}
//合并函数
void unite(int x,int y){
    int rx=find(x);
    int ry=find(y);
    if(rx==ry)return ;
    if(sz[rx]<sz[ry])swap(rx,ry);//保证rx是大树的根
    fa[ry]=rx;sz[rx]+=sz[ry];
}
int main(){
    int N,M;cin>>N>>M;
    fa.resize(N+1);
    sz.assign(N+1,1);
    for(int i=1;i<=N;i++)fa[i]=i;
    for(int i=0;i<M;i++){
        int z,x,y;
        cin>>z>>x>>y;
        if(z==1)unite(x,y);
        else {
            if(find(x)!=find(y)){cout<<"N"<<endl;}
            else {cout<<"Y"<<endl;}
        }
    }
    return 0;
    return 0;
}
