




#include<bits/stdc++.h>
using namespace std;
const int MAXN = 30010;
int fa[MAXN], d[MAXN], sz[MAXN];

int find(int x)
{
    if(fa[x] != x)
    {
        int root = find(fa[x]);
        d[x] += d[fa[x]];
        fa[x] = root;
    }
    return fa[x];
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    // 初始化
    for(int i = 1; i <= 30000; i++)
    {
        fa[i] = i;
        d[i] = 0;
        sz[i] = 1;
    }
    int T;
    cin >> T;
    while(T--)
    {
        char op;
        int x, y;
        cin >> op >> x >> y;
        if(op == 'M')
        {
            int rx = find(x);
            int ry = find(y);
            if(rx != ry)
            {
                fa[rx] = ry;
                d[rx] = sz[ry];
                sz[ry] += sz[rx];
            }
        }
        else
        {
            int rx = find(x);
            int ry = find(y);
            if(rx != ry)
                cout << "-1\n";
            else
                cout << abs(d[x] - d[y]) - 1 << '\n';
        }
    }
    return 0;
}
