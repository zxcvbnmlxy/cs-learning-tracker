// P1892 [BalticOI 2003] 团伙（Day 2）— 扩展域（反集）并查集
// 题解：反集思想，n 个人开 2n 个节点：
//   1 ~ n     真实世界（人本身）
//   n+1 ~ 2n  镜像世界（"我的敌人"聚集地）
// 规则：
//   F x y 朋友 → merge(x, y)            （真实世界合并）
//   E x y 敌人 → merge(x, y+n); merge(y, x+n)
//                （把对方扔进我的镜像 = 对方是我的敌人）
// "敌人的敌人是朋友"由并查集传递性自动保证，无需额外逻辑。
// 最后统计 1~n 里不同根的数量 = 团伙数。

#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2005;   // 2 * 1000，容纳两个世界
int fa[MAXN];

int find(int x)
{
    if (fa[x] != x) fa[x] = find(fa[x]);   // 路径压缩
    return fa[x];
}

void merge(int a, int b)
{
    int ra = find(a), rb = find(b);
    if (ra != rb) fa[ra] = rb;
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= 2 * n; i++) fa[i] = i;

    while (m--)
    {
        char op;
        int x, y;
        cin >> op >> x >> y;
        if (op == 'F')
            merge(x, y);            // 朋友：真实世界合并
        else
        {
            merge(x, y + n);        // 敌人：y 是 x 的敌人
            merge(y, x + n);        //       x 是 y 的敌人
        }
    }

    int ans = 0;
    bool vis[MAXN] = {false};
    for (int i = 1; i <= n; i++)
    {
        int r = find(i);
        if (!vis[r]) { vis[r] = true; ans++; }
    }
    cout << ans << '\n';
    return 0;
}
