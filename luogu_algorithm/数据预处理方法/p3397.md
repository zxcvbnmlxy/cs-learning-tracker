题目描述：在nxn的格子上有m个地毯，给出每一个地毯的(x1,y1)(x2,y2)问每一个点被多少个地毯覆盖？

思路：二维差分，依旧是行和列分别一维差分；
diff[x1][y1]+=v;
diff[x1][y2+1]-=v;
diff[x2+1][y1]-=v;
diff[x2+1][y2+1]+=v;

分别对行和列求前缀和；
for(int i=1;i<=n;i++)
    for(int j=1;j<=n;j++)        
    diff[i][j]+=diff[i-1][j];
for(int i=1;i<=n;i++)
    for(int j=1;j<=n;j++)
    diff[i][j]+=diff[i][j-1];


代码：
#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,m;cin>>n>>m;
    vector<vector<int>>diff(n+2,vector<int>(n+2,0));
    for(int i=0;i<m;i++){
        int x1,y1,x2,y2;
        cin>>x1>>y1>>x2>>y2;
        diff[x1][y1]++;
        diff[x2+1][y1]--;
        diff[x1][y2+1]--;
        diff[x2+1][y2+1]++;
    }
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++)diff[i][j]+=diff[i-1][j];
    }
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++)diff[i][j]+=diff[i][j-1];
    }
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++)cout<<diff[i][j]<<" ";
        cout<<"\n";
    }
    return 0;
}
