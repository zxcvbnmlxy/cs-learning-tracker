
题目描述：两个公司A,B,A公司都是男的（编号都是正的），小明在里面,B公司都是女的（编号都是负数）,小红在里面,然后小明和小红认识，小明（编号1）,小红（编号-1）;
A公司一共N个人,P对朋友关系,B公司一共M个人,Q对朋友关系,朋友的朋友是朋友,任意两个男女都可能成为情侣;
一共最多有多少对情侣（包括小红小明）;

给出N M P Q


思路：并查集，两个集合男女集合求两个中最小的那一个就是答案；











//题目要求的是能够有多少对情侣，一对一，而不是有多少组合！！！！靠

#include<bits/stdc++.h>
using namespace std;
unordered_map<int,int>fa;//A公司
unordered_map<int,int>Fa;//B公司
unordered_map<int,int>sz;//A
unordered_map<int,int>Sz;//B
int find(int x){
    if(!fa.count(x)){
        fa[x]=x;
        sz[x]=1;
    }
    if(fa[x]!=x)fa[x]=find(fa[x]);
    return fa[x];
}
int Find(int x){
    if(!Fa.count(x)){
        Fa[x]=x;
        Sz[x]=1;
    }
    if(Fa[x]!=x)Fa[x]=Find(Fa[x]);
    return Fa[x];
}
void unite(int x,int y){
    int rx=find(x);
    int ry=find(y);
    if(rx==ry)return ;
    if(sz[rx]<sz[ry])swap(rx,ry);
    fa[ry]=rx;sz[rx]+=sz[ry];
}
void Unite(int x,int y){
    int rx=Find(x);
    int ry=Find(y);
    if(rx==ry)return ;
    if(Sz[rx]<Sz[ry])swap(rx,ry);
    Fa[ry]=rx;Sz[rx]+=Sz[ry];
}
int main(){
    int N,M,P,Q;
    cin>>N>>M>>P>>Q;
    for(int i=0;i<P;i++){
        int x,y;
        cin>>x>>y;
        unite(x,y);
    }
    for(int i=0;i<Q;i++){
        int x,y;cin>>x>>y;
        Unite(x,y);
    }
    int rootBoy=find(1);
    int rootGirl=Find(-1);
    long long ans=min(sz[rootBoy],Sz[rootGirl]);
    cout<<ans<<"\n";
    return 0;
}
