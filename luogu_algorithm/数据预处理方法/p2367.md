一维差分 用于解决某个区间内增减数值时使用，核心思路就是等差数列，通过不同的公差实现，就比如2 3 4 5 6
    区间[1,3]增加3 那么在diff[1]+=3,diff[3+1]-=3,然后，用个cur_diff记录当前元素的增减情况，cur_diff+=diff[i],   diff数组大小必须是n+1,防止越界，然后当前元素+=cur_diff


读入 n, p 及初始成绩 a[1..n]。

维护差分数组 diff（初始全 0）。

对每次操作 (x, y, z)：

diff[x] += z

diff[y+1] -= z

对 diff 求前缀和，得到每个位置的总增量 add[i]。

最终成绩 score[i] = a[i] + add[i]，取最小值输出。


#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,p;
    cin>>n>>p;
    vector<long long>a(n+1);
    for(int i=1;i<=n;i++)cin>>a[i];
    vector<long long>diff(n+2,0);
    for(int i=0;i<p;i++){
        int x,y,z;
        cin>>x>>y>>z;
        diff[x]+=z;
        diff[y+1]-=z;
    }
    long long mi=INT_MAX,cur=0;
    for(int i=1;i<=n;i++){
         cur += diff[i];          // cur 即为第 i 个学生的总增量
        long long score = a[i] + cur;
        if (score < mi) mi = score;
    }
    cout<<mi;
    return 0;
}
