# 复杂度

- 时间复杂度: $O(n)$
- 空间复杂度: $O(n)$



# Code
```C++ []
class Solution {
public:
    bool canReach(string s, int minJump, int maxJump) {
        int n=s.size();
        vector<bool>dp(n,false);
        if(s[n-1]=='1')return false;
        dp[0]=true;
        vector<int>pre(n+1,0);
        pre[1]=1;
        for(int j=0;j<n;j++){
            if(s[j]=='0'){
                int left=max(j-maxJump,0);
                int right=j-minJump;
                if(right>=0){
                    int cnt=pre[right+1]-pre[left];
                    if(cnt>0){
                        dp[j]=true;
                        }
                    }
                }
                pre[j+1]=pre[j]+(dp[j]?1:0);
            }
        
        return dp[n-1];
    }
};
```
