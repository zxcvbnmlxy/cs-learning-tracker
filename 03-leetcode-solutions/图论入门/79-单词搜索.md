
> Problem: [79. 单词搜索](https://leetcode.cn/problems/word-search/description/)

# 解题过程


# 复杂度

- 时间复杂度: $O(*)$
- 空间复杂度: $O(*)$



# Code
```C++ []
class Solution {
public:
    int m,n;
    vector<pair<int,int>> dir={{-1,0},{1,0},{0,-1},{0,1}};

    bool dfs(vector<vector<char>>& b, string& w, int x, int y, int idx, vector<vector<bool>>& vis){
        if(idx == w.size()) return true;
        for(auto& d:dir){
            int nx=x+d.first, ny=y+d.second;
            if(nx>=0&&nx<m&&ny>=0&&ny<n && !vis[nx][ny] && b[nx][ny]==w[idx]){
                vis[nx][ny]=true;
                if(dfs(b,w,nx,ny,idx+1,vis)) return true;
                vis[nx][ny]=false;
            }
        }
        return false;
    }

    bool exist(vector<vector<char>>& board, string word) {
        m=board.size(),n=board[0].size();
        for(int i=0;i<m;i++)
            for(int j=0;j<n;j++)
                if(board[i][j]==word[0]){
                    vector<vector<bool>> vis(m,vector<bool>(n,false));
                    vis[i][j]=true;
                    if(dfs(board,word,i,j,1,vis)) return true;
                }
        return false;
    }
};
```
