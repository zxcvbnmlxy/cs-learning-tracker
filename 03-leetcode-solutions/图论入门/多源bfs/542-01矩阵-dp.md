# 思路

动态规划

# 解题过程
当前元素1距离最近的0距离为周围四个点中最小值+1
即dp[i][j]=min{dp[i-1][j],dp[i][j-1],dp[i+1][j],dp[i][j+1]};
遍历dp[i][j]可以从左上和右下算得来，所以遍历顺序为左到右上到下，右到左下到上
初始化只有元素为零的元素距离0的距离为0其余设为最大值


# 复杂度

- 时间复杂度: $O(*)$
- 空间复杂度: $O(*)$

class Solution {
public:
    vector<vector<int>> updateMatrix(vector<vector<int>>& mat) {
        int m=mat.size(),n=mat[0].size();
        vector<vector<int>>dp(m,vector<int>(n,100000));
       for(int i=0;i<m;i++){
        for(int j=0;j<n;j++){
            if(mat[i][j]==0)dp[i][j]=0;
            else {
                if(i>0)dp[i][j]=min(dp[i][j],dp[i-1][j]+1);
                if(j>0)dp[i][j]=min(dp[i][j],dp[i][j-1]+1);
            }
        }
       }
       for(int i=m-1;i>=0;i--){
        for(int j=n-1;j>=0;j--){
            if(mat[i][j]!=0){
                if(i<m-1)dp[i][j]=min(dp[i][j],dp[i+1][j]+1);
                if(j<n-1)dp[i][j]=min(dp[i][j],dp[i][j+1]+1);
            }
        }
       }
       return dp;
    }
};
