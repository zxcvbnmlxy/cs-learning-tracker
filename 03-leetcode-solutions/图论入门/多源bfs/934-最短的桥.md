[934. 最短的桥](https://leetcode.cn/problems/shortest-bridge/description/)

# 思路
BFS

# 复杂度

- 时间复杂度: $O(n*n)$
- 空间复杂度: $O(n*n)$

# Code

class Solution {
public:
    int shortestBridge(vector<vector<int>>& grid) {
        vector<pair<int,int>>dir={{-1,0},{0,-1},{1,0},{0,1}};
        vector<pair<int,int>>island;
        int n=grid.size();
        int ans=0;
        queue<pair<int,int>>q;
        int land_size=0;
        for(int i=0;i<n;i++){
            for(int j=0;j<n;j++){
                if(grid[i][j]==1){
                    q.push({i,j});
                    island.push_back({i,j});
                    grid[i][j]=2;
                    while(!q.empty()){
                        auto [x,y]=q.front();
                        q.pop();
                        for(auto [dx,dy]:dir){
                            int nx=x+dx;
                            int ny=y+dy;
                            if(nx>=0&&nx<n&&ny>=0&&ny<n&&grid[nx][ny]==1){
                                q.push({nx,ny});
                                island.push_back({nx,ny});
                                grid[nx][ny]=2;
                            }
                        }
                    }
                   for(auto [x,y]:island){
                    q.push({x,y});
                   }
                   while(!q.empty()){
                    land_size=q.size();
                    for(int i=0;i<land_size;i++){
                        auto [x,y]=q.front();
                        q.pop();
                        for(auto [dx,dy]:dir){
                            int nx=x+dx;
                            int ny=y+dy;
                            if(nx>=0&&nx<n&&ny>=0&&ny<n){
                            if(grid[nx][ny]==0){
                                q.push({nx,ny});
                                grid[nx][ny]=2; }
                                else if(grid[nx][ny]==1)return ans;
                            } 
                        }
                        
                    }
                   ans++;
                   }
                }
                
            }
        }
        return 0;
    }
};
