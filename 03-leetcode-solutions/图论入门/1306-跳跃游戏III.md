class Solution {
public:
    bool canReach(vector<int>& arr, int start) {
    int n=arr.size();
    stack<int>st;
    st.push(start);
    while(!st.empty()){
        int i=st.top();
        st.pop();
        if(arr[i]==0)return true;
        if(arr[i]==-1)continue;
        int step=arr[i];
        arr[i]=-1;
        int next1=step+i;
        int next2=i-step;
        if(next1>=0&&next1<n)st.push(next1);
        if(next2>=0&&next2<n)st.push(next2);
        }
    return false;
    }
};
