#include<bits/stdc++.h>
using namespace std;
int main(){
    string s;cin>>s;
    long long x,y,n=0;
    stack<long long>stk;
    for(int i=0;i<(int)s.size()-1;i++){
        char ch=s[i];
        switch(ch){
            case'@':break;
            case'+':x=stk.top();stk.pop();y=stk.top();stk.pop();stk.push(x+y);break;
            case'*':x=stk.top();stk.pop();y=stk.top();stk.pop();stk.push(x*y);break;
            case'-':x=stk.top();stk.pop();y=stk.top();stk.pop();stk.push(y-x);break;
            case'/':x=stk.top();stk.pop();y=stk.top();stk.pop();stk.push(y/x);break;
            case'.':stk.push(n);n=0;break;
            default:n=n*10+s[i]-'0';
        }
    }
    cout<<stk.top();
}
