#include<bits/stdc++.h>
using namespace std;
int N;
string M;
int charToval(char ch){
    if(ch<='9')return ch-'0';
    return ch-'A'+10;
}
char valTochar(int x){
    if(x<=9)return x+'0';
    return x-10+'A';
}
bool is(string s){
    int l=0,r=s.size()-1;
    while(l<=r&&s[l]==s[r]){
        l++;r--;
    }
    return l>=r;
}
string cal(string s1,string s2){
    int len1=s1.size()-1,len2=s2.size()-1;
    int carry=0;
    string res;
    while(len1>=0||len2>=0){
        int pos1=(len1>=0)?charToval(s1[len1--]):0;
        int pos2=(len2>=0)?charToval(s2[len2--]):0;
        int sum=pos1+pos2+carry;
        int v=sum%N;
        carry=sum/N;
        res+=valTochar(v);
    }
    if(carry>0)res+=valTochar(carry);
    reverse(res.begin(),res.end());
    return res;
}
int main(){
    cin>>N>>M;
    if(is(M)){cout<<"STEP=0";return 0;}
    for(int step=1;step<=30;step++){
        string pre=M;
        reverse(M.begin(),M.end());
        M=cal(pre,M);
        if(is(M)){cout<<"STEP="<<step;return 0;}
    }
    cout<<"Impossible!";
    return 0;
}
